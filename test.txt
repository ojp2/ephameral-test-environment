
gcloud asset search-all-iam-policies \

  --format=json > iam_policies.json

# Flatten to CSV

  ["member","role","resource"],
  (.[] | .policy.bindings[] as $b | $b.members[] as $m | [$m, $b.role, .resource])
  | @csv
' iam_policies.json > gcp_iam_listing.csv

# ci-ephemeral-pvc-rbac.yaml
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: ci-ephemeral-pvc
  namespace: harness-delegate-ng
rules:
  - apiGroups: [""]
    resources: ["persistentvolumeclaims"]
    verbs: ["get", "list", "watch", "create", "delete"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: ci-ephemeral-pvc
  namespace: harness-delegate-ng
subjects:
  - kind: ServiceAccount
    name: pfc-delegate
    namespace: harness-delegate-ng
roleRef:
  kind: Role
  name: ci-ephemeral-pvc
  apiGroup: rbac.authorization.k8s.io

kubectl apply -f - <<'EOF'
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: cmk-test
spec:
  accessModes: ["ReadWriteOnce"]
  storageClassName: ebs-gp3
  resources:
    requests:
      storage: 1Gi
---
apiVersion: v1
kind: Pod
metadata:
  name: cmk-test
spec:
  containers:
    - name: app
      image: public.ecr.aws/docker/library/busybox:latest
      command: ["sh", "-c", "echo hello > /data/test && sleep 3600"]
      volumeMounts:
        - name: vol
          mountPath: /data
  volumes:
    - name: vol
      persistentVolumeClaim:
        claimName: cmk-test
EOF

# watch it bind + the pod come up
kubectl get pvc cmk-test -w        # STATUS should go Pending -> Bound
kubectl get pod cmk-test           # should reach Running


VOL=$(kubectl get pvc cmk-test -o jsonpath='{.spec.volumeName}')
aws ec2 describe-volumes \
  --filters "Name=tag:kubernetes.io/created-for/pv/name,Values=$VOL" \
  --query 'Volumes[].{Encrypted:Encrypted,KmsKeyId:KmsKeyId}' --output table

kubectl delete pod cmk-test
kubectl delete pvc cmk-test   # reclaimPolicy: Delete means the EBS volume is removed too
