kubectl apply -f - <<'EOF'
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: cmk-test
spec:
  accessModes: ["ReadWriteOnce"]
  storageClassName: ebs-gp3
  resources:
    requests:
      storage: 1Gi
---
apiVersion: v1
kind: Pod
metadata:
  name: cmk-test
spec:
  containers:
    - name: app
      image: public.ecr.aws/docker/library/busybox:latest
      command: ["sh", "-c", "echo hello > /data/test && sleep 3600"]
      volumeMounts:
        - name: vol
          mountPath: /data
  volumes:
    - name: vol
      persistentVolumeClaim:
        claimName: cmk-test
EOF

# watch it bind + the pod come up
kubectl get pvc cmk-test -w        # STATUS should go Pending -> Bound
kubectl get pod cmk-test           # should reach Running


VOL=$(kubectl get pvc cmk-test -o jsonpath='{.spec.volumeName}')
aws ec2 describe-volumes \
  --filters "Name=tag:kubernetes.io/created-for/pv/name,Values=$VOL" \
  --query 'Volumes[].{Encrypted:Encrypted,KmsKeyId:KmsKeyId}' --output table

kubectl delete pod cmk-test
kubectl delete pvc cmk-test   # reclaimPolicy: Delete means the EBS volume is removed too
